    Main upsert logic. Use the Supabase client's `.upsert()` with `onConflict: 'name'` for idempotency. The treasury schema is exposed via PostgREST under the `treasury` schema. Mirror how `bulkLoadGateway.js` and `bulkLoadTransactions.js` access tables — most current scripts use RPC, but for a simple table write we use `supabase.schema('treasury').from('data_sources').upsert(...)`:

    ```javascript
    async function main() {
      console.log(`\nSeeding Dallas data_sources rows (municipality_id ${DALLAS_MUNICIPALITY_ID})...\n`);

      for (const src of SOURCES) {
        const { data, error } = await supabase
          .schema('treasury')
          .from('data_sources')
          .upsert(src, { onConflict: 'name' })
          .select();

        if (error) {
          console.error(`  Error upserting ${src.name}: ${error.message}`);
          process.exit(1);
        }

        const row = Array.isArray(data) ? data[0] : data;
        console.log(`  Upserted: ${src.name}`);
        console.log(`    id            : ${row?.id}`);
        console.log(`    api_type      : ${row?.api_type}`);
        console.log(`    dataset_type  : ${row?.dataset_type}`);
        console.log(`    dataset_id    : ${row?.dataset_id}`);
        console.log(`    fiscal_years  : ${JSON.stringify(row?.fiscal_years)}`);
        console.log('');
      }

      // Verification: list via the same RPC the loader will use
      const { data: listing, error: listErr } = await supabase.rpc('treasury_list_source_ids');
      if (listErr) {
        console.error(`Verification failed (treasury_list_source_ids): ${listErr.message}`);
        process.exit(1);
      }

      const dallasRows = (listing || []).filter(s =>
        s.api_type === 'socrata' &&
        ['operating', 'revenue'].includes(s.dataset_type) &&
        (s.name === 'Dallas Operating Budget' || s.name === 'Dallas Revenue Budget')
      );

      console.log(`Verification: treasury_list_source_ids returns ${dallasRows.length} Dallas budget source(s).`);
      for (const r of dallasRows) {
        console.log(`  - ${r.name} (${r.dataset_type}) FYs: ${(r.fiscal_years || []).join(', ')}`);
      }

      if (dallasRows.length !== 2) {
        console.error(`Expected 2 Dallas budget rows in treasury_list_source_ids, found ${dallasRows.length}`);
        process.exit(1);
      }

      console.log('\nDone. Both Dallas data_sources rows are present and discoverable.\n');
    }

    main().catch(err => { console.error('Fatal:', err); process.exit(1); });
    ```

    CRITICAL pitfalls to honor (from RESEARCH.md):

    - DO NOT hardcode the Supabase URL only — fall back to env var first (matches the existing pattern in bulkLoadTransactions.js line 23).
    - DO NOT inline SQL — use the Supabase JS client. Keeps RLS and schema config intact.
    - The `column_mapping` JSON keys must match EXACTLY what Plan 05-02's loader will read: `fiscal_year_column`, `approved_amount_column`, `actual_amount_column`, `category_column`, `subcategory_column`, `fund_column`. Do not rename or omit any key.
